function generateOBSLink() { const baseUrl = window.location.origin + window.location.pathname; // Keep payload lean: only include properties that differ from defaults to save space const statePayload = { t: widgetState.title, m: widgetState.milestones.map(m => ({ t: m.target, title: m.title })), gs: widgetState.gradStart, ge: widgetState.gradEnd, sc: widgetState.strikeColor, bg: widgetState.cardBg, f: widgetState.fontFamily, h: widgetState.height, sr: widgetState.showRaised ? 1 : 0, sb: widgetState.showBar ? 1 : 0, sm: widgetState.showMilestones ? 1 : 0, ca: widgetState.currentAmount, cen: widgetState.cashAppEnabled ? 1 : 0, ctag: widgetState.cashAppTag, ctxt: widgetState.cashAppText, // Omit custom base64 QR strings from URL if too massive, or compress them cqr: widgetState.customQRData && widgetState.customQRData.length < 500 ? widgetState.customQRData : null }; let encoded = ''; const jsonStr = JSON.stringify(statePayload); // Compress strongly using LZ-String if (typeof LZString !== 'undefined') { encoded = 'lz_' + LZString.compressToEncodedURIComponent(jsonStr); } else { encoded = btoa(encodeURIComponent(jsonStr)); } const obsUrl = `${baseUrl}?obs=true&cfg=${encoded}`; const inputEl = document.getElementById('obs-generated-url'); if (inputEl) inputEl.value = obsUrl; return obsUrl; } function parseURLParams() { const urlParams = new URLSearchParams(window.location.search); if (urlParams.get('obs') === 'true') { document.body.classList.add('obs-mode'); const controlPanel = document.getElementById('control-panel'); if (controlPanel) controlPanel.classList.add('hidden'); const wrapper = document.getElementById('widget-wrapper'); if (wrapper) { wrapper.classList.remove('max-w-xl'); wrapper.classList.add('w-full', 'p-2'); } } const cfgData = urlParams.get('cfg'); if (cfgData) { try { let jsonStr = ''; if (cfgData.startsWith('lz_') && typeof LZString !== 'undefined') { jsonStr = LZString.decompressFromEncodedURIComponent(cfgData.substring(3)); } else { jsonStr = decodeURIComponent(atob(cfgData)); } const decoded = JSON.parse(jsonStr); if (decoded.t) widgetState.title = decoded.t; if (decoded.gs) widgetState.gradStart = decoded.gs; if (decoded.ge) widgetState.gradEnd = decoded.ge; if (decoded.sc) widgetState.strikeColor = decoded.sc; if (decoded.bg) widgetState.cardBg = decoded.bg; if (decoded.f) widgetState.fontFamily = decoded.f; if (decoded.h) widgetState.height = decoded.h; if (decoded.sr !== undefined) widgetState.showRaised = decoded.sr === 1 || decoded.sr === true; if (decoded.sb !== undefined) widgetState.showBar = decoded.sb === 1 || decoded.sb === true; if (decoded.sm !== undefined) widgetState.showMilestones = decoded.sm === 1 || decoded.sm === true; if (decoded.ca !== undefined) widgetState.currentAmount = decoded.ca; if (decoded.cen !== undefined) widgetState.cashAppEnabled = decoded.cen === 1 || decoded.cen === true; if (decoded.ctag) widgetState.cashAppTag = decoded.ctag; if (decoded.ctxt) widgetState.cashAppText = decoded.ctxt; if (decoded.cqr) widgetState.customQRData = decoded.cqr; if (Array.isArray(decoded.m)) { widgetState.milestones = decoded.m.map((item, idx) => ({ id: idx + 1, target: item.t, title: item.title, reached: false })); } } catch (e) { console.error("Failed to parse URL config payload", e); } } }
